from dataclasses import dataclass, field from typing import List, Tuple import math # --- Hilfsfunktionen --- Vec2 = Tuple[float, float] def distance(a: Vec2, b: Vec2) -> float: return math.hypot(a[0] - b[0], a[1] - b[1]) # --- Game-Klassen --- @dataclass class Unit: name: str position: Vec2 hp: int = 100 is_enemy: bool = False def move_to(self, target: Vec2, log): log(f"[{self.name}] bewegt sich zu {target}") self.position = target def take_damage(self, amount: int, log): self.hp -= amount log(f"[{self.name}] erleidet {amount} Schaden (HP: {self.hp})") if self.hp <= 0: log(f"[{self.name}] wurde zerstört.") @dataclass class Drone(Unit): def orbit(self, center: Vec2, radius: float, angle_deg: float, log): rad = math.radians(angle_deg) x = center[0] + math.cos(rad) * radius y = center[1] + math.sin(rad) * radius log(f"[{self.name}] geht in Orbit‑Position bei ({x:.1f}, {y:.1f})") self.position = (x, y) @dataclass class Commander: name: str position: Vec2 units: List[Unit] = log(f"[{self.name}] führt Scan‑Ping bei {center} (R={radius}) aus") detected = [e for e in enemies if distance(e.position, center) <= radius] if not detected: log(" → Keine Feinde im Scanbereich.") for e in detected: log(f" → Feind entdeckt: {e.name} bei {e.position}") return detected def set_drone_orbit(self, radius: float, log): log(f"[{self.name}] setzt Drohnen in Orbit (R={radius})") if not self.drones: log(" Keine Drohnen verfügbar.") return angle_step = 360 / len(self.drones) for i, d in enumerate(self.drones): angle = i * angle_step d.orbit(self.position, radius, angle, log) def orbital_strike(self, enemies: List[Unit], target: Vec2, radius: float, damage: int, log): if self.orbital_cooldown > 0: log(f"[{self.name}] Orbitalangriff noch auf Cooldown: {self.orbital_cooldown} Züge") return log(f"[{self.name}] ruft Himmelslanze auf Ziel {target} (R={radius})") hit_any = False for e in enemies: if distance(e.position, target) <= radius: e.take_damage(damage, log) hit_any = True if not hit_any: log(" → Kein Ziel im Einschlagsradius.") self.orbital_cooldown = 3 def tick(self, log): if self.orbital_cooldown > 0: self.orbital_cooldown -= 1 log(f"[{self.name}] Nächster Zug. Orbital‑Cooldown: {self.orbital_cooldown}") # --- Simple "Engine" für die Browser-Demo --- class CommandHorizonDemo: def __init__(self, log_func, status_func): self.log = log_func self.status = status_func self.reset() def reset(self): self.log("=== NEUE MISSION: FALL DER SKYLINE (DEMO) ===") friendly1 = Unit("Infanterie‑Trupp Alpha", (0, 0)) friendly2 = Unit("Infanterie‑Trupp Beta", (2, 1)) drone1 = Drone("Falcon‑Drohne 1", (0, -2)) drone2 = Drone("Falcon‑Drohne 2", (1, -2)) self.enemies = [ Unit("Dominion‑Drohne A", (10, 10), is_enemy=True), Unit("Dominion‑Drohne B", (12, 9), is_enemy=True), Unit("Dominion‑Mech", (15, 15), hp=200, is_enemy=True), ] self.commander = Commander( name="Kommandant Auron Voss", position=(0, 0), units=[friendly1, friendly2], drones=[drone1, drone2], ) self.status("Mission aktiv.
