import pygame import sys import random pygame.init() # Fenster WIDTH, HEIGHT = 800, 400 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("Red vs Black – Star Battle") clock = pygame.time.Clock() # Farben RED = (255, 0, 0) BLACK = (0, 0, 0) YELLOW = (255, 255, 0) WHITE = (255, 255, 255) GREY = (80, 80, 80) # Spielfeld FLOOR = 350 GRAVITY = 0.6 # Spieler player = pygame.Rect(80, FLOOR - 30, 30, 30) player_vx = 0 player_vy = 0 player_speed = 4 on_ground = True lives = 3 score = 0 # Gegner enemy = pygame.Rect(650, FLOOR - 30, 30, 30) enemy_vx = -2 # Sterne des Spielers stars = [] # Schüsse des Gegners enemy_bullets = [] enemy_shoot_timer = 0 def shoot_player(direction): stars.append({ "x": player.centerx, "y": player.centery, "vx": direction * 7, "r": 6 }) def enemy_shoot(): enemy_bullets.append({ "x": enemy.centerx, "y": enemy.centery, "vx": -5, "r": 5 }) def draw_star(x, y, r): points = [] for i in range(5): angle = i * (3.14159 * 2) / 5 - 3.14159 / 2 x1 = x + r * pygame.math.Vector2(1, 0).rotate_rad(angle).x y1 = y + r * pygame.math.Vector2(1, 0).rotate_rad(angle).y points.append((x1, y1)) pygame.draw.polygon(screen, YELLOW, points) def circle_rect_collision(cx, cy, r, rect): closest_x = max(rect.left, min(cx, rect.right)) closest_y = max(rect.top, min(cy, rect.bottom)) dx = cx - closest_x dy = cy - closest_y return dx*dx + dy*dy < r*r game_over = False while True: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit() sys.exit() keys = pygame.key.get_pressed() if not game_over: # Bewegung player_vx = (keys[pygame.K_d] - keys[pygame.K_a]) * player_speed if (keys[pygame.K_w] or keys[pygame.K_SPACE]) and on_ground: player_vy = -12 on_ground = False # Schießen if keys[pygame.K_j]: shoot_player(1) if keys[pygame.K_k]: shoot_player(-1) # Physik player.x += player_vx player_vy += GRAVITY player.y += player_vy if player.y >= FLOOR - player.height: player.y = FLOOR - player.height player_vy = 0 on_ground = True # Gegner bewegen enemy.x += enemy_vx if enemy.x < 400 or enemy.x > 760: enemy_vx *= -1 # Gegner schießt alle 60 Frames enemy_shoot_timer += 1 if enemy_shoot_timer >= 60: enemy_shoot() enemy_shoot_timer = 0 # Kollision Spieler–Gegner if player.colliderect(enemy): lives -= 1 enemy.x = 650 enemy_vx = random.choice([-2, 2]) if lives <= 0: game_over = True # Sterne bewegen for s in stars: s["x"] += s["vx"] stars = [s for s in stars if -20 < s["x"] < WIDTH + 20] # Sterne treffen Gegner for s in stars[:]: if circle_rect_collision(s["x"], s["y"], s["r"], enemy): score += 1 stars.remove(s) enemy.x = 650 enemy_vx = random.choice([-2, 2]) # Gegner-Schüsse bewegen for b in enemy_bullets: b["x"] += b["vx"] enemy_bullets = [b for b in enemy_bullets if b["x"] > -20] # Gegner-Schüsse treffen Spieler for b in enemy_bullets[:]: if circle_rect_collision(b["x"], b["y"], b["r"], player): lives -= 1 enemy_bullets.remove(b) if lives <= 0: game_over = True # Zeichnen screen.fill((30, 30, 30)) pygame.draw.rect(screen, GREY, (0, FLOOR, WIDTH, 50)) pygame.draw.rect(screen, RED, player) pygame.draw.rect(screen, BLACK, enemy) for s in stars: draw_star(s["x"], s["y"], s["r"]) for b in enemy_bullets: pygame.draw.circle(screen, RED, (int(b["x"]), int(b["y"])), b["r"]) font = pygame.font.SysFont(None, 30) screen.blit(font.render(f"Score: {score}", True, WHITE), (10, 10)) screen.blit(font.render(f"Leben: {lives}", True, WHITE), (10, 40)) if game_over: big = pygame.font.SysFont(None, 60) screen.blit(big.render("GAME OVER", True, WHITE), (280, 150)) pygame.display.flip() clock.tick(60)
